const _memo = new Map();
const memo = (k, f) => { let v = _memo.get(k); if (!v) { v = f(); _memo.set(k, v); } return v; };
// exponentially decaying sine partial, added in place (cheap recursive envelope)
function addPartial(y, f, a, tau, off = 0) {
  if (f >= SR * 0.45) return y;
  const w = TAU * f / SR, k = Math.exp(-1 / (tau * SR));
  let e = a;
  for (let i = off; i < y.length; i++) { y[i] += e * Math.sin(w * (i - off)); e *= k; if (e < 1e-5) break; }
  return y;
}
// stereo SFX with a balance pan
function putS(t, [L, R], g, p = 0, rev = 0.2) {
  const gl = Math.min(1, 1 - p), gr = Math.min(1, 1 + p);
  put(SFX, t, L.map((v) => v * gl), { gain: g, rev, right: R.map((v) => v * gr) });
}

/* glassy in-key bell (addPartial skips partials above Nyquist, so no aliasing on high notes) */
function bell(f, dur = 0.6) {
  return memo('bell' + f.toFixed(2) + '|' + dur, () => {
    const y = new Float32Array(secs(dur));
    addPartial(y, f, 1, dur * 0.28); addPartial(y, f * 2.76, 0.35, dur * 0.08); addPartial(y, f * 5.4, 0.15, dur * 0.04);
    for (let i = 0; i < 40; i++) y[i] *= i / 40;
    return fadeEdges(norm(y), 0.0005, 0.02);
  });
}
function sparkle(t, m, g, p) {   // SFX ting in E-minor pentatonic + octave shimmer
  put(SFX, t, bell(m2f(m), 0.6), { gain: g, pan: p, rev: 0.45 });
  put(SFX, t, bell(m2f(m + 12) * 1.003, 0.25), { gain: g * 0.25, pan: -p, rev: 0.45 });
}
/* Rhodes / electric piano: 1:1 FM body whose index falls with time (the bark), 14:1 tine ping, soft tanh, low-pass */
function rhodes(midi, vel, durS) {
  const d = Math.round(durS * 100) / 100, v = Math.round(vel * 20) / 20;
  return memo('rh' + midi + '|' + v + '|' + d, () => {
    const f = m2f(midi), rel = 0.22, n = secs(d + rel), y = new Float32Array(n);
    const dec = clamp(1.6 * Math.pow(262 / f, 0.6), 0.5, 3.5);
    let p = 0, q = 0;
    for (let i = 0; i < n; i++) {
      const t = i / SR;
      p += f / SR; p -= Math.floor(p); q += 14 * f / SR; q -= Math.floor(q);
      const I = (0.4 + 1.5 * v) * Math.exp(-t / 0.22) + 0.2, tine = 0.9 * v * Math.exp(-t / 0.01);
      const s = 0.75 * Math.sin(TAU * p + I * Math.sin(TAU * p) + tine * Math.sin(TAU * q)) + 0.35 * Math.sin(TAU * p);
      let env = Math.min(1, t / 0.0015) * (0.7 * Math.exp(-t / dec) + 0.3 * Math.exp(-t / (dec * 4)));
      if (t > d) env *= 0.5 + 0.5 * Math.cos(Math.PI * clamp((t - d) / rel, 0, 1));
      y[i] = Math.tanh(1.3 * s) * env;
    }
    return fadeEdges(norm(filt(y, 'lp', 2600 + 2500 * v, 0.7)), 0.0005, 0.03);
  });
}

function tranhLine(b0, notes, g, o = {}) {
  const pan = o.pan ?? 0.2;
  for (const [b, m, d] of notes) {
    const t = BT(b0 + b);
    if (d >= 1 && o.grace !== false) pluck(t - 0.06, pentUp(m), 0.15, g * 0.32, pan);   // luyến: grace from the note above

    if (o.bell) put(MUSIC, t, bell(m2f(m + 12), 0.9), { gain: o.bell, pan: -pan, rev: 0.45 });
  }
}

function shakerHit(v, acc) {
  return memo('shk' + v + acc, () => {
    const R = rngFor('shk' + v + acc), n = secs(0.12), x = noise(n, R), at = 0.012, dc = acc ? 0.04 : 0.024;
    for (let i = 0; i < n; i++) { const t = i / SR; x[i] *= t < at ? Math.pow(t / at, 2) : Math.exp(-(t - at) / dc); }

}
// side-chain: every kick writes a dip into PUMP; the sub-bass bus is multiplied by it at the end of the score
const PUMP = new Float32Array(N).fill(1);
function kickAt(t, g) {
  put(MUSIC, t, KICK_H, { gain: g, rev: 0.03 });
  const i0 = Math.round(t * SR), m = secs(0.4);
  for (let k = 0; k < m; k++) {
    const i = i0 + k; if (i >= N) break;
    const tt = k / SR, d = tt < 0.006 ? smooth(tt / 0.006) : Math.exp(-(tt - 0.006) / 0.11), v = 1 - 0.85 * d;
    if (v < PUMP[i]) PUMP[i] = v;
  }
}

/* the band on a 16th grid (q = position in the 4-beat bar). st: kick (gain scale), clap, hats 8|16, shaker,
 * bass 'house'|'stab'|'long', keys 'house'|'stab'|'pad', pad (warm wavetable pad, pumped when the kick plays), level */
function groove(b0, b1, st) {
  const L = st.level ?? 1;
  for (let q16 = Math.round(b0 * 4); q16 < Math.round(b1 * 4); q16++) {
    const b = q16 / 4, t = BT(b), q = q16 % 16, chN = chordAt(b);
    if (!chN) continue;
    const ch = CH[chN], head = q === 0 || isChange(b) || q16 === Math.round(b0 * 4);
    if (st.kick && q % 4 === 0) kickAt(t, 0.46 * st.kick * L);
    if (st.clap && (q === 4 || q === 12)) put(MUSIC, t, CLAPS[(q16 >> 3) & 1], { gain: 0.15 * L, pan: 0.04, rev: 0.22 });
    if (st.hats) {
      if (q % 4 === 2) put(MUSIC, t, OHAT, { gain: 0.05 * L, pan: 0.22, rev: 0.08 });
      else if (st.hats === 16 || q % 2 === 0) put(MUSIC, t, CHAT[q16 % 3], { gain: (q % 4 === 0 ? 0.032 : 0.022) * L, pan: 0.3, rev: 0.05 });
    }
    if (st.shaker && q % 2 === 1) put(MUSIC, t, shakerHit(q16 % 5, q % 4 === 3), { gain: (q % 4 === 3 ? 0.03 : 0.02) * L, pan: -0.35, rev: 0.05 });
    if (st.bass) {
      const r = ch.r, nx = CH[chordAt(b + 0.5) || chN].r;
      let ev = null;
      if (st.bass === 'house') ev = { 0: [r, 0.45, 1], 3: [r, 0.2, 0.55], 6: [r + 12, 0.3, 0.5], 8: [r, 0.45, 0.9], 10: [r, 0.3, 0.65], 14: [nx !== r ? pentDown(nx) : r + 7, 0.35, 0.6] }[q];
      else if (st.bass === 'stab') ev = { 0: [r, 0.25, 0.9], 10: [r, 0.25, 0.7] }[q];
      else if (st.bass === 'long' && head) ev = [r, segEnd(b) - b - 0.08, 0.85];
      if (ev) bassAt(t, ev[0], BT(ev[1]), ev[2] * L);
    }
    if (st.keys === 'house') {
      if (q === 0) keys(t, chN, 0.6, BT(1.4), 0.062 * L);
      else if (q === 6) keys(t, chN, 0.45, BT(0.35), 0.045 * L);
      else if (q === 10) keys(t, chN, 0.5, BT(0.9), 0.05 * L);
    } else if (st.keys === 'stab' && (q === 2 || q === 10)) keys(t, chN, 0.5, BT(0.3), 0.055 * L);
    else if (st.keys === 'pad' && head) keys(t, chN, 0.5, BT(segEnd(b) - b) - 0.05, 0.06 * L, { roll: 0.012 });
    if (st.pad && head) {
      const t0 = t, d = BT(segEnd(b) - b) - 0.04;
      put2(MUSIC, t0, pad(ch.k, d, { attack: 0.25, release: 0.3, cutoff: st.padCut ?? 1000, pump: st.kick ? (lt) => pumpAt(t0 + lt) : null }), { gain: 0.03 * L, rev: 0.3 });
    }
  }
}

/* lo-fi vinyl: sparse crackle, a few pops, band-limited hiss with a 33 rpm wobble */
function vinyl(R, dur, fadeOut = 0.6) {
  const n = secs(dur), y = norm(crackle(R, dur, () => 22, () => 1, 900, 4500));
  for (let t = R(); t < dur - 0.05; t += 0.6 + R() * 1.5) addInto(y, norm(burst(R, 0.01, 'lp', 2500, 0.7, 0.0015)), secs(t), 1.4);

/* the egg's flight: slide-whistle arc G5 -> G6 -> B5 + air */
function slideArc(R, dur) {
  const n = secs(dur), y = new Float32Array(n), f0 = m2f(79), f1 = m2f(91), f2 = m2f(83); let ph = 0;
  for (let i = 0; i < n; i++) {
    const t = i / SR, u = t / dur, f = u < 0.45 ? expi(f0, f1, smooth(u / 0.45)) : expi(f1, f2, smooth((u - 0.45) / 0.55));
    ph += f * (1 + 0.012 * Math.sin(TAU * 6 * t)) / SR; ph -= Math.floor(ph);
    y[i] = (Math.sin(TAU * ph) + 0.12 * Math.sin(2 * TAU * ph)) * smooth(u / 0.08) * smooth((1 - u) / 0.15);
  }
  const w = whoosh(R, dur, 400, 2400, 900, 0.45, 1.0);
  for (let i = 0; i < n; i++) y[i] = y[i] * 0.45 + w[i] * 0.55;
  return fadeEdges(norm(y), 0.005, 0.02);
}
/* tiny cat 'mrrp' chirp (voiced, rising, trilled) + paw thup */
function catChirp(R) {
  const dur = 0.24, n = secs(dur), src = new Float32Array(n); let ph = 0;
  for (let i = 0; i < n; i++) {
    const t = i / SR, u = t / dur, f = pw([[0, 560], [0.35, 820], [1, 700]], u);
    ph += f / SR; ph -= Math.floor(ph);
    let s = 0; for (let h = 1; h <= 8; h++) s += Math.sin(TAU * h * ph) / h;
    src[i] = s * smooth(t / 0.015) * Math.pow(1 - u, 0.8) * (t < 0.07 ? 0.55 + 0.45 * Math.sin(TAU * 32 * t) : 1);
  }
  const y = new Float32Array(secs(0.32));
  addInto(y, filt(src, 'bp', 1300, 3), 0, 1); addInto(y, filt(src, 'bp', 2900, 5), 0, 0.5);
  addInto(y, plop(R, 180, 90, 0.08), 0, 0.15);
  return fadeEdges(norm(filt(y, 'hp', 300, 0.7)), 0.002, 0.03);
}

/* THE BITE: lips meet the sticky glaze, the tender egg gives (juicy pop + squelch), crispy garlic in the glaze,
 * a puff of steam, a short slurp of tamarind sauce. Stereo; gentle rounding only (no heavy saturation). */
function biteEgg(R, scale = 1) {
  const n = secs(1.1), L = new Float32Array(n), Rr = new Float32Array(n);
  addPan(L, Rr, norm(burst(R, 0.012, 'bp', 2200, 0.8, 0.0025)), 0, 0.5, 0);                     // 1) sticky smack
  addPan(L, Rr, plop(R, 270, 95, 0.32), secs(0.004), 0.7, 0);                                    // 2) the egg gives: juicy pop
  const o1 = secs(0.006), sqL = norm(squish(rngFor('biteSqL' + scale), 0.55)), sqR = norm(squish(rngFor('biteSqR' + scale), 0.55));
  const wL = norm(filt(noise(sqL.length, rngFor('biteWetL' + scale)), 'bp', (t) => 1500 + 500 * Math.sin(TAU * 4.3 * t), 1.3));
  const wR = norm(filt(noise(sqL.length, rngFor('biteWetR' + scale)), 'bp', (t) => 1700 + 500 * Math.sin(TAU * 3.7 * t + 1), 1.3));
  for (let i = 0; i < sqL.length && i + o1 < n; i++) {                                          // 3) wet squelch (~0.35 s dense)
    const t = i / SR, e = Math.min(1, t / 0.006) * Math.exp(-t / 0.17);
    L[i + o1] += (sqL[i] * 0.6 + wL[i] * 0.45) * e; Rr[i + o1] += (sqR[i] * 0.6 + wR[i] * 0.45) * e;
  }
  for (let t = 0; t < 0.28;) {                                                                    // 4) crispy garlic / chilli bits
    const a = 0.4 * Math.exp(-t / 0.12) * (0.4 + 0.6 * R());
    addPan(L, Rr, burst(R, 0.012, 'bp', 1500 + R() * 4500, 1.2, 0.0015 + R() * 0.004), secs(t), a, (R() - 0.5) * 1.6);
    t += -Math.log(1 - R() * 0.99) / (700 * Math.exp(-t / 0.1) + 25);
  }
  const stL = whoosh(rngFor('biteStL' + scale), 0.6, 1500, 5000, 2500, 0.15, 0.7), stR = whoosh(rngFor('biteStR' + scale), 0.6, 1600, 5400, 2300, 0.18, 0.7);
  addInto(L, stL, secs(0.02), 0.16); addInto(Rr, stR, secs(0.02), 0.16);                           // 5) puff of steam
  const sl = norm(filt(noise(secs(0.25), R), 'bp', (t) => expi(700, 2400, t / 0.25), 3));
  for (let i = 0; i < sl.length; i++) sl[i] *= Math.sin(Math.PI * i / sl.length);
  addPan(L, Rr, sl, secs(0.3), 0.22, 0);                                                          // 6) tamarind slurp
  norm2(L, Rr);
  for (let i = 0; i < n; i++) { L[i] = Math.tanh(1.2 * L[i]) / Math.tanh(1.2); Rr[i] = Math.tanh(1.2 * Rr[i]) / Math.tanh(1.2); }
  fadeEdges(L, 0.0005, 0.06); fadeEdges(Rr, 0.0005, 0.06);
  return norm2(L, Rr);
}

/* low lub-dub thump (tanh for audible harmonics on phones) */
function thump(R, f = 52, dec = 0.09) {
  const y = new Float32Array(secs(0.35)); let ph = 0;
  for (let i = 0; i < y.length; i++) { const t = i / SR; ph += (f + 30 * Math.exp(-t / 0.02)) / SR; y[i] = Math.tanh(1.8 * Math.sin(TAU * ph)) * Math.min(1, t / 0.003) * Math.exp(-t / dec); }
  addInto(y, norm(burst(R, 0.06, 'lp', 300, 0.8, 0.015)), 0, 0.3);
  return fadeEdges(norm(y), 0.0005, 0.03);
}
/* the spoon trembles: faint metallic rattle, crescendo */
function tremble(R, dur) {
  const n = secs(dur), y = new Float32Array(n);
  addPartial(y, 2630, 0.6, 2); addPartial(y, 3910, 0.4, 2); addPartial(y, 5480, 0.25, 2);
  const nz = norm(filt(noise(n, R), 'bp', 3200, 2));
  for (let i = 0; i < n; i++) {
    const t = i / SR, u = t / dur, am = Math.pow(Math.abs(Math.sin(TAU * 7 * t + 0.6 * Math.sin(TAU * 1.3 * t))), 4);
    y[i] = (y[i] * 0.6 + nz[i] * 0.4) * am * smooth(u / 0.4) * smooth((1 - u) / 0.1);
  }
  return fadeEdges(norm(y), 0.01, 0.03);
}
