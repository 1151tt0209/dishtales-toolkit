const _memo = new Map();
const memo = (k, f) => { let v = _memo.get(k); if (!v) { v = f(); _memo.set(k, v); } return v; };
// exponentially decaying sine partial, added in place (cheap recursive envelope)
function addPartial(y, f, a, tau, off = 0) {
  if (f >= SR * 0.45) return y;
  const w = TAU * f / SR, k = Math.exp(-1 / (tau * SR));
  let e = a;
  for (let i = off; i < y.length; i++) { y[i] += e * Math.sin(w * (i - off)); e *= k; if (e < 1e-5) break; }
  return y;
}

/* marimba bar: modal synthesis (1 : 3.93 : 9.2 overtones of a tuned bar), soft rubber mallet */
function marimbaNote(midi, vel = 1) {
  return memo('mar' + midi + '|' + vel, () => {
    const f = m2f(midi), R = rngFor('mar' + midi + '|' + vel);
    const d1 = clamp(0.6 * Math.pow(262 / f, 0.55), 0.14, 1.5);
    const y = new Float32Array(secs(Math.min(3.2, d1 * 5)));
    addPartial(y, f, 1, d1);
    addPartial(y, f * 3.932, 0.28 * vel, d1 * 0.2);
    addPartial(y, f * 9.21, 0.08 * vel * vel, d1 * 0.07);
    for (let i = 0; i < 60; i++) y[i] *= i / 60;
    addInto(y, norm(burst(R, 0.02, 'lp', 1400 + 2200 * vel, 0.7, 0.0025)), 0, 0.07 * vel);
    return fadeEdges(norm(y), 0.0005, 0.03);
  });
}
/* glockenspiel: bright steel bar (1 : 2.756 : 5.404 : 8.933) */
function glockNote(midi, dur = 1.8) {
  return memo('gl' + midi + '|' + dur, () => {
    const f = m2f(midi), R = rngFor('gl' + midi);
    const y = new Float32Array(secs(dur));
    addPartial(y, f, 1, dur * 0.3);
    addPartial(y, f * 2.756, 0.2, 0.12);
    addPartial(y, f * 5.404, 0.07, 0.05);
    addPartial(y, f * 8.933, 0.025, 0.025);
    addInto(y, norm(burst(R, 0.006, 'hp', 5000, 0.7, 0.0012)), 0, 0.06);
    return fadeEdges(norm(y), 0.0004, 0.05);
  });
}
/* ukulele string: Karplus–Strong nylon pluck with a small body (430 Hz / 2.4 kHz) */
function ukeString(midi, v = 0) {
  return memo('uke' + midi + '|' + v, () => {
    const R = rngFor('uke' + midi + '|' + v), f = m2f(midi);
    const P = Math.max(2, Math.round(SR / f - 0.5)), buf = new Float32Array(P);
    let lp = 0; for (let i = 0; i < P; i++) { lp += 0.45 * ((R() * 2 - 1) - lp); buf[i] = lp; }
    let mean = 0; for (let i = 0; i < P; i++) mean += buf[i]; mean /= P; for (let i = 0; i < P; i++) buf[i] -= mean;
    const decay = Math.exp(-1 / (0.55 * f)), n = secs(1.5), y = new Float32Array(n);
    let idx = 0;
    for (let i = 0; i < n; i++) { const a = buf[idx], b = buf[(idx + 1) % P]; y[i] = a; buf[idx] = decay * 0.5 * (a + b); idx = (idx + 1) % P; }
    let z = filt(y, 'peak', 430, 1.3, 5); z = filt(z, 'peak', 2400, 1.2, 2); z = filt(z, 'hp', 110, 0.7);
    return fadeEdges(norm(z), 0.0005, 0.2);
  });
}
const ukeMute = (m, v) => memo('ukm' + m + '|' + v, () => {
  const s = Float32Array.from(ukeString(m, v).subarray(0, secs(0.1)));
  for (let i = 0; i < s.length; i++) s[i] *= Math.exp(-i / SR / 0.03);
  return fadeEdges(s, 0, 0.01);
});
/* upright bass: plucked, warm harmonics decaying faster upward, finger thump */
function upright(midi, durS) {
  return memo('ub' + midi + '|' + durS.toFixed(3), () => {
    const f = m2f(midi), R = rngFor('ub' + midi), rel = 0.06, n = secs(durS + rel), y = new Float32Array(n);
    const amps = [1, 0.55, 0.32, 0.2, 0.12, 0.07], E = amps.slice(), K = amps.map((_, h) => Math.exp(-1 / ((1.1 / (1 + 0.7 * h)) * SR)));
    let ph = 0;
    for (let i = 0; i < n; i++) {
      const t = i / SR;
      ph += f * (1 + 0.01 * Math.exp(-t / 0.025)) / SR; if (ph > 1e6) ph -= 1e6;
      let s = 0; for (let h = 0; h < 6; h++) { s += E[h] * Math.sin(TAU * (h + 1) * ph); E[h] *= K[h]; }
      let env = Math.min(1, t / 0.004);
      if (t > durS) env *= 0.5 + 0.5 * Math.cos(Math.PI * clamp((t - durS) / rel, 0, 1));
      y[i] = s * env;
    }
    addInto(y, norm(burst(R, 0.05, 'lp', 260, 0.8, 0.012)), 0, 0.3);
    return fadeEdges(norm(filt(y, 'lp', 1400, 0.7)), 0.001, 0.01);
  });
}
function shakerHit(v, acc) {
  return memo('shk' + v + acc, () => {
    const R = rngFor('shk' + v + acc), n = secs(0.12), x = noise(n, R), at = 0.012, dc = acc ? 0.04 : 0.024;
    for (let i = 0; i < n; i++) { const t = i / SR; x[i] *= t < at ? Math.pow(t / at, 2) : Math.exp(-(t - at) / dc); }
    return fadeEdges(norm(filt(filt(x, 'hp', 4300, 0.7), 'peak', 8500, 1, 5)), 0.0005, 0.01);
  });
}
function rimHit(v) {

    const R = rngFor('rim' + v), y = new Float32Array(secs(0.15));
    addPartial(y, 1720, 0.6, 0.011); addPartial(y, 520, 0.5, 0.022); addPartial(y, 3100, 0.2, 0.006);
    addInto(y, norm(burst(R, 0.006, 'hp', 3000, 0.7, 0.0012)), 0, 0.5);
    return fadeEdges(norm(y), 0.0003, 0.02);
  });
}

function strum(t, chN, o = {}) {
  const notes = CH[chN].u, up = o.dir === 'u', g = o.gain ?? 0.06, sp = o.spread ?? 0.012, pan0 = o.pan ?? -0.32;
  (up ? [3, 2, 1, 0] : [0, 1, 2, 3]).forEach((si, k) => {
    const m = notes[si], v = (Math.round(t * 13) + si) % 3;
    put(MUSIC, t + k * sp, o.mute ? ukeMute(m, v) : ukeString(m, v), { gain: g * (up ? 0.7 : 1) * (1 - 0.05 * k), pan: pan0 + (si - 1.5) * 0.09, rev: o.rev ?? 0.2 });
  });
}
/* the band, on a 16th grid. st: kick '13'|'four', ghost, rim, shaker 8|16, bass 'folk'|'stac', uke 'strum'|'soft'|'chop', arps 8|16, level */
function groove(b0, b1, st) {
  const L = st.level ?? 1;
  for (let q16 = Math.round(b0 * 4); q16 < Math.round(b1 * 4); q16++) {
    const b = q16 / 4, t = BT(b), q = q16 % 16, chN = chordAt(b);
    if (!chN) continue;
    const ch = CH[chN];
    if (st.kick === 'four' ? q % 4 === 0 : st.kick === '13' ? (q === 0 || q === 8) : false) put(MUSIC, t, KICK_S, { gain: 0.42 * L, rev: 0.03 });
    if (st.kick && st.ghost && q === 10) put(MUSIC, t, KICK_S, { gain: 0.2 * L, rev: 0.03 });
    if (st.rim && (q === 4 || q === 12)) put(MUSIC, t, rimHit(q16 % 4), { gain: 0.13 * L, pan: -0.12, rev: 0.18 });
    if (st.shaker && (st.shaker === 16 || q % 2 === 0)) {
      const acc = q % 4 === 2;
      put(MUSIC, t, shakerHit(q16 % 5, acc), { gain: (acc ? 0.075 : (q % 2 ? 0.035 : 0.05)) * L, pan: 0.38, rev: 0.06 });
    }
    if (st.bass) {
      const r = ch.r, fifth = r + 7 <= 50 ? r + 7 : r - 5, nx = CH[chordAt(b + 0.5) || chN].r;
      let ev = null;
      if (st.bass === 'folk') ev = { 0: [r, 0.75], 6: [fifth, 0.35], 8: [r, 0.75], 12: [fifth, 0.35], 14: [nx !== r ? stepBelow(nx) : r + 12, 0.25] }[q];
      else ev = { 0: [r, 0.2], 4: [fifth, 0.2], 8: [r + 12, 0.2], 12: [fifth, 0.2] }[q];
      if (ev) put(MUSIC, t, upright(ev[0], BT(ev[1])), { gain: 0.34 * L * (q === 0 ? 1 : 0.85), pan: 0.02 });
    }
    if (st.uke === 'strum') { const d = { 0: 'd', 4: 'd', 6: 'u', 10: 'u', 12: 'd', 14: 'u' }[q]; if (d) strum(t, chN, { dir: d, gain: (q === 0 ? 0.075 : 0.06) * L }); }
    else if (st.uke === 'soft' && (q === 0 || q === 8)) strum(t, chN, { gain: 0.05 * L, spread: 0.025 });
    else if (st.uke === 'chop' && (q === 4 || q === 12)) strum(t, chN, { gain: 0.08 * L, mute: true });
    if (st.arps === 8 && q % 2 === 0) {
      const idx = [0, 2, 1, 3, 2, 1, 3, 2][q / 2], acc = q === 0 || q === 6 || q === 12;   // 3+3+2 tresillo accents
      put(MUSIC, t, marimbaNote(ch.t[idx], acc ? 0.9 : 0.6), { gain: (acc ? 0.08 : 0.055) * L, pan: 0.32, rev: 0.25 });
    }
    if (st.arps === 16) {
      const idx = [0, 1, 2, 3, 2, 1, 2, 3, 0, 1, 2, 3, 2, 3, 2, 1][q];
      put(MUSIC, t, marimbaNote(ch.t[idx] + 12, q % 4 === 0 ? 0.8 : 0.5), { gain: (q % 4 === 0 ? 0.055 : 0.036) * L, pan: q % 2 ? 0.42 : 0.22, rev: 0.25 });
    }
  }
}
/* melody: marimba lead (rolled on long notes) + optional glockenspiel double */
function melody(b0, notes, o = {}) {
  const pan = o.pan ?? -0.12;
  for (const [b, m, d] of notes) {
    const t = BT(b0 + b);
    if (o.mar) {
      put(MUSIC, t, marimbaNote(m, 1), { gain: o.mar, pan, rev: 0.3 });
      if (d >= 1.5) for (let k = 1, tt = t + 0.085; tt < t + BT(d) - 0.07; k++, tt += 0.085)
        put(MUSIC, tt, marimbaNote(m, 0.5), { gain: o.mar * (0.4 - 0.15 * (tt - t) / BT(d)), pan: pan + (k % 2 ? 0.06 : -0.06), rev: 0.3 });
    }
    if (o.glock) put(MUSIC, t, glockNote(m + (o.gOct ?? 12)), { gain: o.glock, pan: o.gPan ?? 0.24, rev: 0.4 });
  }
}

/* cloth twisted over a bowl: fabric creak + wet squish + trickle */
function clothSqueeze(R, dur) {
  const n = secs(dur), y = new Float32Array(n), sq = norm(squish(R, dur));
  const imp = new Float32Array(n);
  for (let t = 0.05; t < dur - 0.05;) { const u = t / dur; imp[secs(t)] = (0.5 + 0.5 * R()) * Math.sin(Math.PI * u); t += (1 / (25 + 30 * u)) * (0.9 + 0.2 * R()); }
  const cr = norm(filt(filt(imp, 'bp', 750, 5), 'peak', 1600, 3, 6));
  const tr = norm(filt(noise(n, R), 'bp', (t) => 2200 + 600 * Math.sin(TAU * 7 * t), 2));
  for (let i = 0; i < n; i++) {
    const t = i / SR, u = t / dur;
    y[i] = sq[i] * 0.55 + cr[i] * 0.4 + tr[i] * 0.22 * smooth((u - 0.35) / 0.2) * smooth((1 - u) / 0.15) * (0.6 + 0.4 * Math.sin(TAU * 13 * t));
  }
  return fadeEdges(norm(y), 0.01, 0.03);
}

/* sieve tapped over the bowl, powdery hiss */
function flourSift(R, dur) {
  const n = secs(dur), y = new Float32Array(n);
  const hiss = norm(filt(filt(noise(n, R), 'bp', 5200, 0.5), 'hp', 2500, 0.7));
  const taps = []; for (let t = 0.02; t < dur - 0.05; t += 0.1 + R() * 0.02) taps.push(t);
  for (let i = 0; i < n; i++) {
    const t = i / SR; let e = 0.25;
    for (const tp of taps) if (t >= tp) e += 0.8 * Math.exp(-(t - tp) / 0.045);
    y[i] = hiss[i] * 0.5 * e * smooth(t / 0.04) * smooth((dur - t) / 0.12);
  }
  for (const tp of taps) addInto(y, norm(burst(R, 0.01, 'bp', 1900 + R() * 600, 2.5, 0.003)), secs(tp), 0.3);
  addInto(y, norm(crackle(R, dur, () => 300, (t) => smooth((dur - t) / 0.15), 4000, 9000)), 0, 0.2);
  return fadeEdges(norm(y), 0.005, 0.02);
}

  const cl = (t0, len, rate0, g) => {
    for (let t = 0; t < len;) {
      const a = g * Math.exp(-t / (len * 0.45)) * (0.4 + 0.6 * R());
      addPan(L, Rr, burst(R, 0.016, 'bp', 600 + R() * 3000, 1.1, 0.002 + R() * 0.006), secs(t0 + t), a, (R() - 0.5) * 1.6);
      t += -Math.log(1 - R() * 0.99) / (rate0 * Math.exp(-t / (len * 0.4)) + 25);
    }
  };
  cl(0, 0.3, 900, 0.55);                                       // 1) thin golden crust: warm, mid-range crackle
  const pres = norm(filt(noise(secs(0.3), R), 'bp', 2300, 0.6));
  for (let i = 0; i < pres.length; i++) { const t = i / SR; pres[i] *= Math.min(1, t / 0.002) * Math.exp(-t / 0.05); }
  addPan(L, Rr, pres, 0, 0.33, 0);
  const o1 = secs(0.01), sqL = norm(squish(rngFor('biteSqL'), 0.6)), sqR = norm(squish(rngFor('biteSqR'), 0.6));
  const wL = norm(filt(noise(sqL.length, rngFor('biteWetL')), 'bp', (t) => 1700 + 500 * Math.sin(TAU * 4.3 * t), 1.4)), wR = norm(filt(noise(sqL.length, rngFor('biteWetR')), 'bp', (t) => 1900 + 500 * Math.sin(TAU * 3.7 * t + 1), 1.4));
  for (let i = 0; i < sqL.length; i++) {                      // 2) spongy squelch: low body + wet mid layer, ~0.3 s of dense energy
    const t = i / SR, e = Math.min(1, t / 0.008) * Math.exp(-t / 0.2);
    L[i + o1] += (sqL[i] * 0.6 + wL[i] * 0.35) * e; Rr[i + o1] += (sqR[i] * 0.6 + wR[i] * 0.35) * e;
  }
  const wh = norm(filt(noise(secs(0.4), R), 'bp', 1200, 0.7));
  for (let i = 0; i < wh.length; i++) { const t = i / SR; wh[i] *= Math.min(1, t / 0.004) * Math.exp(-t / 0.09); }
  addPan(L, Rr, wh, 0, 0.55, 0);                               // 3) airy honeycomb collapse
  const th = new Float32Array(secs(0.4)); let ph = 0;
  for (let i = 0; i < th.length; i++) { const t = i / SR; ph += (75 + 50 * Math.exp(-t / 0.025)) / SR; th[i] = Math.sin(TAU * ph) * Math.min(1, t / 0.002) * Math.exp(-t / 0.08); }
  addPan(L, Rr, fadeEdges(th, 0.001, 0.02), 0, 0.35, 0);      // 4) teeth meet: soft low thump (kept small: peaks without loudness)
  addPan(L, Rr, stretchSqueak(R, 0.25, 420, 620, { segs: 1 }), secs(0.32), 0.18, 0);   // 5) sticky pull-away
  cl(0.3, 0.15, 220, 0.35);
  norm2(L, Rr);
  for (let i = 0; i < n; i++) { L[i] = Math.tanh(1.2 * L[i]) / Math.tanh(1.2); Rr[i] = Math.tanh(1.2 * Rr[i]) / Math.tanh(1.2); }   // gentle rounding only
  fadeEdges(L, 0.0005, 0.06); fadeEdges(Rr, 0.0005, 0.06);
  return norm2(L, Rr);
}
