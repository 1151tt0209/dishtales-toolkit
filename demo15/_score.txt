// ───────────────────────────── 6. Score (15 s demo, 108 BPM, banhmi-style: one SFX per on-screen action) ─────────────────────────────
// beat -> seconds helper (shot cuts: 0 / 2.778 / 4.444 / 6.667 / 8.333 / 10.0 / 11.111)
const BT = (b) => b * BEAT;

// ---------- ACT I : hook + groove (0–10.0), hard silence 10.0–10.45, then the bite ----------
{
  const R = rngFor('demoI');
  // warm fire bed under the whole kitchen part
  put2(SFX, 0.0, fireBed(rngFor('fire1'), 10.0, 8), { gain: 0.07, rev: 0.15 });

  // s1: ladle whooshes in, batter hits the hot pan at 0.30 (sizzle + splat), lace bubbles pop as it sets
  put(SFX, 0.0, whoosh(rngFor('ladle'), 0.3, 300, 1800, 500, 0.6, 1.0), { gain: 0.35, pan: (t) => 0.8 - t * 1.6, rev: 0.2 });
  put2(SFX, 0.30, pourSizzle(rngFor('batter'), 1.5), { gain: 0.95, rev: 0.12 });
  put(SFX, 0.30, plop(rngFor('splat'), 200, 70, 0.3), { gain: 0.5, pan: 0, rev: 0.2 });
  [[1.6, 0.25, -0.3], [2.15, 0.3, 0.3]].forEach(([t, g, p], k) => put(SFX, t, crustCrack(rngFor('bub' + k)), { gain: g, pan: p, rev: 0.25 }));

  // s2: each piece lands in the pan on its own plop (3.05 / 3.28 / 3.50 / 3.72 / 3.94), searing sizzle, sprouts rustle
  [[3.05, 230, 85, -0.2], [3.28, 260, 95, 0.2], [3.50, 245, 90, -0.3], [3.72, 255, 95, 0.3], [3.94, 240, 88, 0]].forEach(([t, f0, f1, p], k) =>
    put(SFX, t, plop(rngFor('land' + k), f0, f1, 0.22), { gain: 0.5, pan: p, rev: 0.15 }));
  put2(SFX, 3.05, pourSizzle(rngFor('sear'), 1.2), { gain: 0.55, rev: 0.12 });
  put2(SFX, 4.05, leafRustle(rngFor('sprouts'), 0.5), { gain: 0.6, rev: 0.2 });

  // s3: spatula slides in, lace crackles as it crisps, the fold closes with a crisp crack at 6.0
  put(SFX, 4.9, whoosh(rngFor('spat'), 0.35, 500, 3000, 900, 0.5), { gain: 0.35, pan: 0.3, rev: 0.2 });
  [[5.3, 0.35], [5.55, 0.4], [5.8, 0.35]].forEach(([t, g], k) => put(SFX, t, crustCrack(rngFor('lace' + k)), { gain: g, pan: (k - 1) * 0.4, rev: 0.25 }));
  put2(SFX, 6.0, bigCrack(rngFor('fold')), { gain: 0.75, rev: 0.25 });

  // s4: cut whoosh, plate lands (6.95), glassy tings in key
  put(SFX, 6.35, whoosh(rngFor('cut4'), 0.35, 400, 2500, 700, 0.55, 1.2), { gain: 0.5, pan: (t) => -0.6 + t * 2, rev: 0.2 });
  put(SFX, 6.95, plop(rngFor('plate'), 150, 52, 0.5), { gain: 0.85, pan: 0, rev: 0.3 });
  [[7.25, 98, -0.4], [7.6, 101, 0.4], [7.95, 105, 0]].forEach(([t, m, p]) => {
    put(SFX, t, ting(m2f(m), 0.6), { gain: 0.21, pan: p, rev: 0.45 });
    put(SFX, t, ting(m2f(m + 12) * 1.003, 0.25), { gain: 0.05, pan: -p, rev: 0.45 });
  });

  // s5: leaf rustle while Má Tư rolls (8.4), the dip lands at 9.35, one sauce droplet
  put2(SFX, 8.4, leafRustle(rngFor('leaf'), 0.8), { gain: 0.8, rev: 0.2 });
  put2(SFX, 9.35, bloopSplash(rngFor('dip')), { gain: 0.62, rev: 0.4 });
  put(SFX, 9.62, ting(m2f(105), 0.4), { gain: 0.1, pan: 0.2, rev: 0.45 });

  // ---- music ----
  put(MUSIC, 0.0, drone(10.0, { attack: 1.2, release: 0.3 }), { gain: 0.1, rev: 0.15 });
  const chords = [[0, [50, 57, 62, 65]], [2.222, [53, 57, 60, 65]], [4.444, [48, 55, 60, 65]], [6.667, [50, 57, 62, 65]], [8.889, [55, 62, 67, 72]]];
  chords.forEach(([t, ch], i) => put2(MUSIC, t, pad(ch, 2.2, { attack: i === 0 ? 0.8 : 0.08, release: 0.2, cutoff: 1500, pump: t >= 2.7 ? ((tt) => pumpAt(t + tt)) : undefined }), { gain: 0.07, rev: 0.3 }));
  // đàn tranh roll in the hook
  [62, 69, 74, 77].forEach((m, k) => tranhNote(0.25 + k * 0.08, m, 1.6, 0.07, -0.5 + k * 0.33, 'roll' + k));
  // groove from the first cut (2.778): kick every beat, claps on 2 & 4, 8th hats, bass
  const K = kick();
  for (let t = BT(5); t < CUT - 1e-6; t += BEAT) put(MUSIC, t, K, { gain: 0.5, rev: 0.03 });
  for (let b = 5; b < 18; b++) if ((b - 5) % 2 === 1) put(MUSIC, BT(b), clap(rngFor('clap' + b)), { gain: 0.2, pan: 0.05, rev: 0.22 });
  for (let t = BT(5); t < CUT - 1e-6; t += BEAT / 2) {
    const off = Math.round((t - BT(5)) / (BEAT / 2)) % 2 === 1;
    put(MUSIC, t, hat(rngFor('hat' + t), off ? 0.04 : 0.025), { gain: off ? 0.065 : 0.04, pan: 0.3, rev: 0.05 });
  }
  const bassRoots = [[BT(5), 38], [BT(9), 41], [BT(13), 36], [BT(17), 43]];
  bassRoots.forEach(([T0, root], bi) => {
    [[0, 0, 0.22], [1.5, 0, 0.18], [2, 'f', 0.22], [3, 12, 0.12], [3.5, 'f', 0.2]].forEach(([ob, iv, d]) => {
      const t = T0 + ob * BEAT; if (t >= CUT - 1e-6) return;
      put(MUSIC, t, bassNote(iv === 'f' ? fifthOf(root) : root + iv, d), { gain: 0.3 });
    });
  });
  // cut accents on each visual cut (alternating pan)
  put2(MUSIC, 6.667, crash(rngFor('c4'), 1.2), { gain: 0.07, rev: 0.25 });
  [[BT(5), 'tom'], [BT(8), 'tick'], [BT(15), 'tick']].forEach(([t, k], i) => {
    if (k === 'tom') put(MUSIC, t, tom(i % 2 ? 98 : 123), { gain: 0.16, pan: 0.3, rev: 0.2 });
    else put(MUSIC, t, tick(rngFor('tk' + i)), { gain: 0.07, pan: i % 2 ? 0.45 : -0.45, rev: 0.3 });
  });
  // đàn bầu motifs [beat, midi, dur, bendFrom]
  lead(0.15, 69, 1.4, { bendFrom: -2, bendTime: 0.45, vibDepth: 0.35, attack: 0.12, gain: 0.17 });
  lead(1.9, 72, 0.5, { bendFrom: -1, bendTime: 0.12, gain: 0.15 });
  [[5, 69, 0.22, -2], [5.5, 72, 0.22, 0], [6, 74, 0.45, -2], [8, 72, 0.22, 0], [8.5, 69, 0.22, 0], [9, 67, 0.45, 2],
   [12, 74, 0.45, -2], [13, 72, 0.22, 0], [13.5, 69, 0.22, 0], [14, 69, 0.95, -2], [16, 77, 0.45, -3], [16.5, 74, 0.45, 0], [17.2, 81, 0.7, -5]]
    .forEach(([b, m, d, bf]) => lead(BT(b), m, d, { bendFrom: bf, gain: 0.17, ...(b === 17.2 ? { vibDepth: 0.5, attack: 0.15, rev: 0.6 } : {}) }));
  // đàn tranh 8th ostinato from the first cut, quiet on the right
  const patIdx = [0, 2, 4, 2, 5, 4, 2, 3];
  for (let k = 0; k < 14; k++) {
    const t = BT(5) + k * (BEAT / 2); if (t >= CUT - 0.3) break;
    tranhNote(t, SCALE[Math.min(SCALE.length - 1, patIdx[k % 8])], 0.5, 0.05, 0.55, 'os' + k, 0.35);
  }
  // build into the silence: riser ends exactly at 10.0, snare roll
  put(MUSIC, 9.0, riser(R, 1.0), { gain: 0.22, pan: (t) => 0.3 * Math.sin(TAU * 1.5 * t), rev: 0.3 });
  [[9.2, 0.1], [9.4, 0.12], [9.6, 0.14], [9.72, 0.15], [9.84, 0.17], [9.93, 0.19]].forEach(([t, g], k) => put(MUSIC, t, clap(rngFor('roll' + k)), { gain: g, pan: k % 2 ? 0.15 : -0.15, rev: 0.25 }));
}

// ---------- 10.0–10.45: silence (put() truncation + master gate) ----------

// ---------- ACT II : the bite (10.45) + sunset CTA (11.11–15.0) ----------
{
  // the bite: loudest SFX of the film
  put2(SFX, RESUME, biteCrunch(rngFor('bite')), { gain: 1.5, rev: 0.22 });
  // ringing chord after the bite
  lead(10.5, 74, 1.2, { bendFrom: -7, bendTime: 0.55, vibDepth: 0.5, attack: 0.08, release: 0.15, gain: 0.22, rev: 0.6 });
  lead(10.5, 62, 1.2, { bendFrom: -7, bendTime: 0.55, vibDepth: 0.3, attack: 0.08, release: 0.15, gain: 0.08, rev: 0.6, pan: 0.2 });
  put2(MUSIC, 10.5, pad([50, 57, 62, 67, 69], 1.0, { attack: 0.12, release: 0.15, cutoff: 2000, detune: 8 }), { gain: 0.13, rev: 0.5 });
  put(MUSIC, 10.5, boom(1.4), { gain: 0.38, rev: 0.15 });
  [50, 57, 62, 69, 74].forEach((m, k) => tranhNote(10.5 + k * 0.045, m, 1.4, 0.06, -0.4 + k * 0.2, 'fin' + k, 0.55));

  // CTA groove: kick on every beat from 11.111, claps on 2 & 4, hats, bass, bright tranh, playful bầu
  const K2 = kick();
  for (let b = 20; b < 27; b++) put(MUSIC, BT(b), K2, { gain: 0.5, rev: 0.03 });
  for (let b = 21; b < 27; b += 2) put(MUSIC, BT(b), clap(rngFor('cc' + b)), { gain: 0.2, pan: 0.05, rev: 0.22 });
  for (let t = BT(20); t < DUR - 0.4; t += BEAT / 2) {
    const off = Math.round((t - BT(20)) / (BEAT / 2)) % 2 === 1;
    put(MUSIC, t, hat(rngFor('ch' + t), off ? 0.04 : 0.025), { gain: off ? 0.065 : 0.04, pan: 0.3, rev: 0.05 });
  }
  [[BT(20), 38], [BT(24), 43]].forEach(([T0, root]) => {
    [[0, 0, 0.22], [1.5, 0, 0.18], [2, 'f', 0.22], [3, 12, 0.12]].forEach(([ob, iv, d]) => put(MUSIC, T0 + ob * BEAT, bassNote(iv === 'f' ? fifthOf(root) : root + iv, d), { gain: 0.3 }));
  });
  put2(MUSIC, BT(20), pad([57, 62, 67, 69], 3.0, { attack: 0.1, release: 0.3, cutoff: 2400, pump: (t) => pumpAt(BT(20) + t) }), { gain: 0.085, rev: 0.3 });
  [[20, 74, 0.22, 0], [20.5, 77, 0.22, 0], [21, 79, 0.45, -1], [22.5, 81, 0.22, 0], [23, 79, 0.22, 0], [23.5, 77, 0.45, 0], [24, 74, 0.45, -2]]
    .forEach(([b, m, d, bf]) => lead(BT(b), m, d, { bendFrom: bf, gain: 0.17 }));
  for (let k = 0; k < 12; k++) tranhNote(BT(20) + k * (BEAT / 2), SCALE[Math.min(SCALE.length - 1, 4 + [0, 2, 1, 3, 2, 1][k % 6])], 0.5, 0.055, 0.5, 'cta' + k, 0.35);
  // final hit on beat 26 (14.44), ring out, silent by 15.0
  put2(MUSIC, BT(26), crash(rngFor('endc'), 0.8), { gain: 0.16, rev: 0.3 });
  put(MUSIC, BT(26), boom(0.6), { gain: 0.3, rev: 0.12 });
  [62, 69, 74, 81].forEach((m, k) => tranhNote(BT(26) + k * 0.04, m, 0.7, 0.07, -0.4 + k * 0.26, 'end' + k, 0.5));

  // UI sounds, each on its on-screen moment: like pop 11.67, follow slides 12.0, tap 12.78 + confetti sparkle
  put(SFX, 11.67, plop(rngFor('likepop'), 420, 160, 0.14), { gain: 0.4, pan: -0.3, rev: 0.2 });
  put(SFX, 11.67, ting(m2f(105), 0.35), { gain: 0.14, pan: -0.3, rev: 0.4 });
  put(SFX, 12.0, whoosh(rngFor('fol'), 0.3, 600, 3500, 1000, 0.5), { gain: 0.3, pan: (t) => 0.9 - t * 0.8, rev: 0.2 });
  put(SFX, 12.78, tick(rngFor('tap')), { gain: 0.5, pan: 0.3, rev: 0.15 });
  [[12.78, 98], [12.86, 105], [12.94, 110]].forEach(([t, m], k) => put(SFX, t, ting(m2f(m), 0.5), { gain: 0.16, pan: -0.2 + k * 0.3, rev: 0.45 }));
  put2(SFX, 12.8, crash(rngFor('conf'), 0.8), { gain: 0.06, rev: 0.3 });
}
